--HANDLERS--
function vtStam pName, pS
   return value("VT.Stamina(" & quote & pName & quote & ")", pS)
end vtStam

function vtPart pState, pPart, pItem
   set the itemDelimiter to "|"
   put item pPart of pState into pState
   set the itemDelimiter to comma
   return item pItem of pState
end vtPart
